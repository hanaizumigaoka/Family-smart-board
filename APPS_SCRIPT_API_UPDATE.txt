const API_TOKEN = "THAY_BANG_TOKEN_RIENG_DAI_NGAU_NHIEN";

function doGet(e) {
  const params = (e && e.parameter) ? e.parameter : {};
  const callback = String(params.callback || "");
  let result;
  try {
    if (String(params.token || "") !== API_TOKEN) {
      result = { success: false, message: "Unauthorized" };
    } else if ((params.action || "notices") === "notices") {
      result = getSakuraNoticesData();
    } else {
      result = { success: false, message: "Unknown action" };
    }
  } catch (err) {
    result = { success: false, message: String(err && err.message ? err.message : err) };
  }
  return apiOutput(result, callback);
}

function getSakuraNoticesData() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName("SakuraInbox");
  if (!sheet || sheet.getLastRow() < 2) return { success: true, count: 0, notices: [] };
  const values = sheet.getDataRange().getValues();
  const headers = values[0];
  const notices = values.slice(1).map(row => {
    const item = {};
    headers.forEach((header, index) => {
      if (header === "") return;
      const value = row[index];
      item[String(header)] = value instanceof Date ? value.toISOString() : value;
    });
    return item;
  });
  notices.sort((a, b) => new Date(b["Ngày gửi"] || 0) - new Date(a["Ngày gửi"] || 0));
  return { success: true, count: notices.length, notices: notices };
}

function apiOutput(data, callback) {
  if (callback && /^[A-Za-z_$][0-9A-Za-z_$\.]*$/.test(callback)) {
    return ContentService.createTextOutput(callback + "(" + JSON.stringify(data) + ");")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
